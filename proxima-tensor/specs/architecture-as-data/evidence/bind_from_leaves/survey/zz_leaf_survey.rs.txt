#![cfg(feature = "std")]
#![allow(clippy::unwrap_used, clippy::expect_used)]

use std::collections::{BTreeMap, BTreeSet};
use std::fmt::Write as _;
use std::fs::File;

use proxima_gguf::parse_complete;
use proxima_model_interop::{ArchitectureRegistry, KvLayout};
use proxima_tensor::op::Op;

fn survey(name: &str, path: &str) {
    let file = File::open(path).unwrap();
    let mapping = unsafe { memmap2::Mmap::map(&file) }.unwrap();
    let bytes: &[u8] = &mapping;
    let parsed = parse_complete(bytes).unwrap();
    let registry = ArchitectureRegistry::with_builtin();
    let route = registry.resolve(&parsed).unwrap();
    let bound = route
        .bind_with_kv_layout(&parsed, bytes, KvLayout::SlidingRing)
        .unwrap();
    let mut consumers: BTreeMap<u32, Vec<String>> = BTreeMap::new();
    for op in &bound.program {
        for dependency in op.dependencies() {
            let label = match op {
                Op::Elementwise { body, operands, .. } => {
                    let map = operands
                        .iter()
                        .find(|(node, _)| *node == dependency)
                        .map(|(_, index_map)| format!("{index_map:?}"))
                        .unwrap_or_default();
                    let map: String = map.chars().take(70).collect();
                    format!("ew:{body:?}:{map}")
                }
                Op::Reduce(reduce) => format!(
                    "reduce:{:?}:in={}",
                    reduce.body,
                    format!("{:?}", reduce.in_map).chars().take(70).collect::<String>()
                ),
                _ => "other".to_string(),
            };
            consumers.entry(dependency.0).or_default().push(label);
        }
    }
    let mut kinds: BTreeMap<String, &str> = BTreeMap::new();
    for (tensor, _) in bound.weights.owned() {
        kinds.insert(tensor.clone(), "owned");
    }
    for (tensor, _) in bound.weights.packed() {
        kinds.insert(tensor.clone(), "packed");
    }
    for (tensor, _, _) in bound.weights.packed_owned() {
        kinds.insert(tensor.clone(), "packed_owned");
    }
    let mut out = String::new();
    let mut leaf_names = BTreeSet::new();
    for (index, op) in bound.program.iter().enumerate() {
        if let Op::Input { dtype, shape, name: leaf } = op {
            let leaf = leaf.clone().unwrap_or_default();
            leaf_names.insert(leaf.clone());
            let tensor = parsed.tensors.iter().find(|candidate| candidate.name == leaf);
            let consumer_text = consumers
                .get(&(index as u32))
                .map(|labels| {
                    let mut labels = labels.clone();
                    labels.sort();
                    labels.dedup();
                    labels.join(" | ")
                })
                .unwrap_or_default();
            writeln!(
                out,
                "LEAF\t{leaf}\t{dtype:?}\t{shape:?}\tgguf={}\tbound={}\t{consumer_text}",
                tensor
                    .map(|found| format!("{:?}/{:?}", found.dims.as_slice(), found.ggml_type))
                    .unwrap_or_else(|| "-".into()),
                kinds.get(&leaf).copied().unwrap_or("-")
            )
            .unwrap();
        }
    }
    for (tensor, kind) in &kinds {
        if !leaf_names.contains(tensor) {
            writeln!(out, "BOUND_NOT_LEAF\t{tensor}\t{kind}").unwrap();
        }
    }
    for tensor in &parsed.tensors {
        if !kinds.contains_key(&tensor.name) {
            writeln!(
                out,
                "GGUF_UNBOUND\t{}\t{:?}\t{:?}",
                tensor.name,
                tensor.dims.as_slice(),
                tensor.ggml_type
            )
            .unwrap();
        }
    }
    std::fs::write(
        format!(
            "/Users/brianbruggeman/repos/slot-0/.long_ctx_backups/arch_data/slice_8/survey_{name}.tsv"
        ),
        out,
    )
    .unwrap();
}
